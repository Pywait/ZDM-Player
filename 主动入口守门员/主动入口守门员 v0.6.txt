// ==UserScript==
// @name         主动入口守门员 v0.7
// @namespace    http://tampermonkey.net/
// @version      0.7
// @description  修复重定向失效 + 动态暂停视频
// @match        *://*.douyin.com/*
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    // 记录暂停视频的定时器，点按钮后要能把它关掉
    let pauseInterval = null;

    // ==========================================
    // 模块一：左上角的小标签
    // ==========================================
    function insertTag() {
        if (document.getElementById('gatekeeper-tag')) return;

        const tag = document.createElement('div');
        tag.id = 'gatekeeper-tag';
        tag.textContent = '守门员在线 (v0.7)';
        tag.style.cssText = `
            position: fixed; top: 0px; left: 0px;
            z-index: 2147483647; background: rgb(0, 245, 212);
            font-size: 18px; color: black; padding: 12px 12px;
            border-radius: 6px; font-weight: bold; font-family: sans-serif;
            box-shadow: rgb(0, 245, 212, 0.66) 4px 4px 10px;
        `;
        document.body.appendChild(tag);
    }

    // ==========================================
    // 模块二：女仆意图遮罩面板
    // ==========================================
    function initGatekeeper() {
        // 【v0.7 修改】不再用全局变量拦截，而是检查页面上是不是已经有面板了
        // 这样即使抖音改地址重定向，只要没弹出过面板，它依然会执行
        if (document.getElementById('gatekeeper-wrapper')) return;

        // 1. 全屏遮罩
        const wrapper = document.createElement('div');
        wrapper.id = 'gatekeeper-wrapper';
        wrapper.style.cssText = `
            position: fixed; top: 0; left: 0; width: 100vw; height: 100vh;
            background: rgba(0, 0, 0, 0.7); z-index: 2147483646;
            display: flex; align-items: center; justify-content: center;
            backdrop-filter: blur(5px);
        `;

        // 2. 女仆面板主体
        const panel = document.createElement('div');
        panel.style.cssText = `
            background: #1a1a1a; color: #00f5d4; padding: 40px;
            border-radius: 16px; box-shadow: 0 10px 30px rgba(0, 245, 212, 0.3);
            text-align: center; font-family: sans-serif; max-width: 400px;
        `;

        // 3. 女仆文案
        const text = document.createElement('h2');
        text.innerHTML = '主人，您回来啦~<br><span style="font-size:16px;color:#ccc;font-weight:normal;display:block;margin-top:10px;">今天是想认真看点东西，还是让奴家陪您探索未知？<br>或者……就摆烂一小会儿也行哦。</span>';

        // 4. 按钮容器
        const btnBox = document.createElement('div');
        btnBox.style.cssText = 'display: flex; gap: 10px; justify-content: center; margin-top: 30px;';

        // 5. 三个意图按钮
        const buttons = [
            { text: '认真看', color: '#00f5d4', hover: '#00d2b8' },
            { text: '去探索', color: '#f5a623', hover: '#d18e1e' },
            { text: '摆烂刷', color: '#ff4d4d', hover: '#cc3d3d' }
        ];

        buttons.forEach(item => {
            const btn = document.createElement('button');
            btn.textContent = item.text;
            btn.style.cssText = `
                padding: 10px 20px; border: none; border-radius: 8px;
                background: ${item.color}; color: #000; font-weight: bold;
                cursor: pointer; font-size: 16px; transition: 0.2s;
            `;
            btn.onmouseover = () => btn.style.background = item.hover;
            btn.onmouseout = () => btn.style.background = item.color;

            btn.onclick = () => {
                console.log(`用户选择了：${item.text}`);
                wrapper.remove(); // 删掉遮罩

                // 【v0.7 新增】清理暂停定时器，防止它一直在后台偷偷暂停新滑到的视频
                if (pauseInterval) {
                    clearInterval(pauseInterval);
                    pauseInterval = null;
                }

                // 恢复播放
                const videos = document.querySelectorAll('video');
                videos.forEach(video => {
                    video.play().catch(e => console.log('浏览器拦截了自动播放，需用户手动点击'));
                });
            };

            btnBox.appendChild(btn);
        });

        // 6. 组装并挂载
        panel.appendChild(text);
        panel.appendChild(btnBox);
        wrapper.appendChild(panel);
        document.body.appendChild(wrapper);
        console.log('✅ 女仆面板已挂载！');

        // ==========================================
        // 7. 【v0.7 核心修复】动态暂停视频（解决“暂停 0 个视频”的尴尬）
        // 原理：抖音加载视频很慢，不能只查一次。我们开启一个定时器，每 0.5 秒去查一次，
        // 只要面板还在，查到一个暂停一个。
        // ==========================================
        pauseInterval = setInterval(() => {
            const videos = document.querySelectorAll('video');
            if (videos.length > 0) {
                videos.forEach(video => {
                    // 如果视频正在播放，就暂停它
                    if (!video.paused) {
                        video.pause();
                        console.log('⏸️ 成功拦截一个视频播放');
                    }
                });
            }
        }, 500); // 500 毫秒 = 0.5 秒
    }

    // ==========================================
    // 启动执行 + 监听地址变化
    // ==========================================
    function startGatekeeper() {
        insertTag();
        setTimeout(initGatekeeper, 1500);
    }

    // 【v0.7 新增】监听 URL 变化。
    // 抖音重定向时，地址栏会变，这个事件就会触发，防止我们的脚本被甩掉。
    window.addEventListener('popstate', () => {
        // 如果检测到当前不是推荐流页面，或者已经做过了选择，就不弹
        // 这里为了测试，先只做防重入检查
        setTimeout(initGatekeeper, 1000);
    });

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', startGatekeeper);
    } else {
        startGatekeeper();
    }

})();
